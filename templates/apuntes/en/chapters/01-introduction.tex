\chapter{Introduction}
\label{ch:introduction}

\begin{definition}
  A vector in space is written in the standard basis as
  $\vec{v} = a\,\ihat + b\,\jhat + c\,\khat$.
\end{definition}

\begin{theorem}[Euclidean norm]
  \label{thm:norm}
  For every $\vec{v} = a\,\ihat + b\,\jhat + c\,\khat$,
  \[
    \lVert \vec{v} \rVert = \sqrt{a^2 + b^2 + c^2}.
  \]
\end{theorem}

\begin{proof}
  Apply the Pythagorean theorem twice.
\end{proof}

\begin{example}
  If $\vec{v} = 3\,\ihat + 4\,\jhat$, then $\lVert \vec{v} \rVert = 5$
  (Theorem~\ref{thm:norm}).
\end{example}
